// 属性（値）を持つ区間 [l, r) を管理する構造体
/*
=======================================================================
【区間をsetで管理して上書きするやつ (Chtholly Tree / ODT) の備忘録】

◆ 何ができるか
・区間 [l, r) を値 v で「ベタ塗り（上書き）」する。
・その際、「塗りつぶされて消える古い区間と値」を回収して差分計算ができる。

◆ 必須構造
struct Node {
    T l, r;
    V val;
    bool operator<(const Node& rhs) const { return l < rhs.l; }
};
set<Node> st;

◆ 必須関数1: split(x)
・x を境界として区間を2つに割り、[x, ...) 側のイテレータを返す。

◆ 必須関数2: assign(l, r, val)  ※ここが一番バグる！
・区間 [l, r) を val で上書きし、消えた区間達を返す/処理する。

◆ 初期化・実装のコツ
・必ず【半開区間 [l, r)】で扱うこと。入力が閉区間なら r++ する。
・最初に全体を覆うデカい区間（番兵）を入れておくと、イテレータが end() を
  指してバグる事故を防げる。 ex) st.insert({0, INF, 0});
=======================================================================
*/
template <typename T, typename V>
struct IntervalMap {
    struct Node {
        T l, r;
        V val;
        bool operator<(const Node& rhs) const {
            return l < rhs.l;
        }
    };
    
    set<Node> st;

    IntervalMap(T min_val, T max_val, V default_val) {
        // 番兵を入れるか、初期区間を1つドカンと入れるのが一般的
        st.insert({min_val, max_val, default_val});
    }

    // x を含む区間 [l, r) を [l, x) と [x, r) に分割し、[x, r) のイテレータを返す
    // すでに境界がある場合や、範囲外の場合は適宜イテレータを返す
    auto split(T x) {
        auto it = prev(st.upper_bound({x, x, V()}));
        if (it->l == x) return it; // 既に境界がある
        if (it->r <= x) return next(it); // 完全に右側（番兵等で回避可能だが一応）

        T l = it->l, r = it->r;
        V val = it->val;
        st.erase(it);
        st.insert({l, x, val});
        return st.insert({x, r, val}).first;
    }

    // 区間 [l, r) の値を val に上書き（更新）し、
    // 更新前にあった区間の情報を返す（これを使って差分計算などを行う）
    vector<Node> assign(T l, T r, V val) {
        auto itr = split(r); // 順番が重要：必ず右端(r)から split する！
        auto itl = split(l); // その後左端(l)を split する
        
        vector<Node> erased_nodes;
        for (auto it = itl; it != itr; ++it) {
            erased_nodes.push_back(*it);
        }
        
        st.erase(itl, itr);
        st.insert({l, r, val});
        
        return erased_nodes;
    }

    // デバッグ用
    void dump() const {
        for (const auto& node : st) {
            cout << "[" << node.l << ", " << node.r << ") : " << node.val << endl;
        }
    }
};
