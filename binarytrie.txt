// T: 扱う整数の型 (int, long long)
// MAX_LOG: 最大ビット数 (intなら30, long longなら60)
template <typename T = long long, int MAX_LOG = 60>
struct BinaryTrie {
    static_assert(std::is_integral_v<T> && MAX_LOG > 0 && MAX_LOG <= std::numeric_limits<T>::digits);
    struct Node {
        int nxt[2];
        int count; // この部分木に含まれる要素数
        Node() : count(0) { nxt[0] = nxt[1] = -1; }
    };

    vector<Node> nodes;
    int root;

    BinaryTrie() : root(0) {
        nodes.push_back(Node());
    }

    // 値 val を追加 (デフォルト1個)
    void insert(T val, int k = 1) {
        assert(k >= 0 && val >= 0 && ((std::make_unsigned_t<T>(val) >> (MAX_LOG - 1)) >> 1) == 0);
        int now = root;
        nodes[now].count += k;
        for (int i = MAX_LOG - 1; i >= 0; i--) {
            int b = (val >> i) & 1;
            if (nodes[now].nxt[b] == -1) {
                nodes[now].nxt[b] = nodes.size();
                nodes.push_back(Node());
            }
            now = nodes[now].nxt[b];
            nodes[now].count += k;
        }
    }

    // 値 val を削除 (デフォルト1個)
    // ※ 存在しない値を消そうとすると壊れるので注意
    void erase(T val, int k = 1) {
        if (count_exact(val) < k) return; // 安全対策
        int now = root;
        nodes[now].count -= k;
        for (int i = MAX_LOG - 1; i >= 0; i--) {
            int b = (val >> i) & 1;
            now = nodes[now].nxt[b];
            nodes[now].count -= k;
        }
    }

    // 値 val がいくつ入っているか
    int count_exact(T val) const {
        if (val < 0 || ((std::make_unsigned_t<T>(val) >> (MAX_LOG - 1)) >> 1) != 0) return 0;
        int now = root;
        for (int i = MAX_LOG - 1; i >= 0; i--) {
            int b = (val >> i) & 1;
            if (nodes[now].nxt[b] == -1) return 0;
            now = nodes[now].nxt[b];
        }
        return nodes[now].count;
    }

    // 集合内の全要素を xor_val で XOR したとき、小さい方から k 番目 (0-indexed) の値を返す
    // 戻り値は「XOR した後の値」。元の値が欲しい場合は (戻り値 ^ xor_val) をする
    T get_kth(int k, T xor_val = 0) const {
        assert(0 <= k && k < nodes[root].count);
        assert(xor_val >= 0 && ((std::make_unsigned_t<T>(xor_val) >> (MAX_LOG - 1)) >> 1) == 0);
        int now = root;
        T res = 0;
        for (int i = MAX_LOG - 1; i >= 0; i--) {
            int b = (xor_val >> i) & 1;
            int l = nodes[now].nxt[b];
            int r = nodes[now].nxt[b ^ 1];
            int l_cnt = (l == -1 ? 0 : nodes[l].count);
            
            if (k < l_cnt) {
                now = l;
            } else {
                k -= l_cnt;
                now = r;
                res |= (T(1) << i);
            }
        }
        return res;
    }

    // xor_val で XOR したときの最小値 / 最大値を返す
    T get_min(T xor_val = 0) const { return get_kth(0, xor_val); }
    T get_max(T xor_val = 0) const { return get_kth(nodes[root].count - 1, xor_val); }

    // 集合内の要素のうち、(val ^ x) < limit を満たす val の個数を返す
    int count_less_than(T x, T limit) const {
        assert(x >= 0 && ((std::make_unsigned_t<T>(x) >> (MAX_LOG - 1)) >> 1) == 0);
        if (limit <= 0) return 0;
        if (((std::make_unsigned_t<T>(limit) >> (MAX_LOG - 1)) >> 1) != 0) return size();
        int now = root;
        int res = 0;
        for (int i = MAX_LOG - 1; i >= 0; i--) {
            if (now == -1) break;
            int b_xor = (x >> i) & 1;
            int b_lim = (limit >> i) & 1;
            
            int l = nodes[now].nxt[b_xor];     // XORして 0 になる進み方
            int r = nodes[now].nxt[b_xor ^ 1]; // XORして 1 になる進み方

            if (b_lim == 1) {
                // limit のビットが 1 なら、XORして 0 になる側は全て limit 未満確定なので count を足す
                if (l != -1) res += nodes[l].count;
                now = r; // ギリギリの勝負をしている 1 の側へ潜る
            } else {
                // limit のビットが 0 なら、XORして 0 になる側にしか進めない
                now = l;
            }
        }
        return res;
    }

    // 現在の要素数
    int size() const { return nodes[root].count; }
};
