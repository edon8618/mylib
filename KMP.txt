// --- KMP法の pi配列（Prefix function）を構築する関数 ---
// 引数: 文字列 s
// 返り値: 長さ |s| の vector<int> pi
// pi[i] : s[0..i] における、接頭辞と接尾辞が一致する最大文字数
vector<int> KMP(const string& s) {
    int n = s.size();
    vector<int> pi(n, 0);
    for (int i = 1; i < n; i++) {
        int j = pi[i - 1];
        while (j > 0 && s[i] != s[j]) {
            j = pi[j - 1]; // 失敗辺を辿る
        }
        if (s[i] == s[j]) {
            j++;
        }
        pi[i] = j;
    }
    return pi;
}
