 * push, erase, pop, top, empty, size
 */
// Compare = std::less<T> をデフォルトに設定（標準と同じく最大値が取り出される）
template <typename T, typename Compare = std::less<T>>
class ErasablePriorityQueue {
private:
    // 内部のキューにも Compare を適用する
    std::priority_queue<T, std::vector<T>, Compare> pq;
    std::priority_queue<T, std::vector<T>, Compare> delayed_del;

    void clean() {
        while (!pq.empty() && !delayed_del.empty() && pq.top() == delayed_del.top()) {
            pq.pop();
            delayed_del.pop();
        }
    }

public:
    void push(const T& val) {
        pq.push(val);
    }

    void erase(const T& val) {
        delayed_del.push(val);
    }

    void pop() {
        clean();
        if (!pq.empty()) {
            pq.pop();
        }
    }

    T top() {
        clean();
        return pq.top();
    }

    bool empty() {
        clean();
        return pq.empty();
    }

    size_t size() {
        return pq.size() - delayed_del.size();
    }
};
